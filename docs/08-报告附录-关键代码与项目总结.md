### 1. 数据库访问层 — 连接池与参数化查询

**文件**：`backend/src/config/db.js`  
**作用**：统一管理 SQL Server 连接，所有 SQL 使用参数绑定，防止 SQL 注入。

```javascript
// 数据库连接配置（从 .env 读取服务器、库名、账号等）
const config = {
  server: process.env.DB_SERVER || 'localhost',
  port: parseInt(process.env.DB_PORT || '1433', 10),
  database: process.env.DB_NAME || 'SDO_OA',
  pool: { max: 10, min: 0, idleTimeoutMillis: 30000 }, // 连接池复用
};

let pool = null;

// 单例连接池：避免每次请求重复建连
async function getPool() {
  if (pool) return pool;
  pool = await sql.connect(config);
  return pool;
}

// 参数化查询：@key 与 params 对象绑定，防止 SQL 注入
async function query(text, params = {}) {
  const p = await getPool();
  const request = p.request();
  for (const [key, value] of Object.entries(params)) {
    request.input(key, value);
  }
  return request.query(text);
}
```

### 2. 身份认证中间件 — JWT 校验与角色授权

**文件**：`backend/src/middleware/auth.js`  
**作用**：保护需登录接口；管理端接口叠加角色校验。

```javascript
// 校验请求头中的 Bearer Token，解析后写入 req.user
function requireAuth(req, res, next) {
  const header = req.headers.authorization;
  if (!header || !header.startsWith('Bearer ')) {
    return fail(res, '未登录或 Token 无效', 401, 401);
  }
  try {
    const token = header.slice(7);
    // 验证 JWT 签名与有效期，载荷含 userId、username、role
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch {
    return fail(res, 'Token 已过期或无效', 401, 401);
  }
}

// 角色守卫：仅允许 admin 等指定角色访问
function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user || !roles.includes(req.user.role)) {
      return fail(res, '权限不足', 403, 403);
    }
    next();
  };
}
```

### 3. 用户登录 — bcrypt 密码校验与 JWT 签发

**文件**：`backend/src/routes/auth.js`  
**作用**：完成登录认证，区分账号状态（待审核/已拒绝/正常）。

```javascript
router.post('/login', [...], async (req, res, next) => {
  const { account, password } = req.body;

  // 1. 按用户名或邮箱查询用户
  const result = await query(
    `SELECT user_id, username, email, password_hash, role, status, ...
     FROM Users WHERE username = @account OR email = @account`,
    { account }
  );
  const user = result.recordset[0];
  if (!user) return fail(res, '用户名或密码错误', 1, 401);

  // 2. bcrypt 比对哈希密码（不明文存储）
  const ok = await bcrypt.compare(password, user.password_hash);
  if (!ok) return fail(res, '用户名或密码错误', 1, 401);

  // 3. 账号状态校验
  if (user.status === 'pending') return fail(res, '账号待审核', 1, 403);
  if (user.status === 'rejected') return fail(res, '账号已被拒绝', 1, 403);

  // 4. 签发 JWT，前端后续请求携带
  const token = jwt.sign(
    { userId: user.user_id, username: user.username, role: user.role },
    process.env.JWT_SECRET,
    { expiresIn: process.env.JWT_EXPIRES_IN || '24h' }
  );
  delete user.password_hash; // 响应中不返回哈希
  return success(res, { token, user }, '登录成功');
});
```

### 4. 每日打卡 — 幂等写入

**文件**：`backend/src/routes/dashboard.js`  
**作用**：成员每日打卡一次；数据库唯一约束 `(user_id, clock_date)` 保证不重复。

```javascript
router.post('/clock-in', requireAuth, async (req, res, next) => {
  const userId = req.user.userId;

  // 查重：同一用户同一自然日只能打卡一次
  const existing = await query(
    `SELECT record_id FROM ClockInRecords
     WHERE user_id = @userId AND clock_date = CAST(GETDATE() AS DATE)`,
    { userId }
  );
  if (existing.recordset.length) {
    return fail(res, '今日已打卡', 1, 400);
  }

  // 写入打卡记录，clock_date 取服务器当天日期
  const result = await query(
    `INSERT INTO ClockInRecords (user_id, clock_date, clock_in_time)
     OUTPUT INSERTED.clock_in_time
     VALUES (@userId, CAST(GETDATE() AS DATE), GETDATE())`,
    { userId }
  );
  return success(res, { clockedIn: true, clockInTime: result.recordset[0].clock_in_time }, '打卡成功');
});
```

### 5. 课程视频 — MP4 同步与演示列表

**文件**：`backend/src/utils/videoPrepare.js`  
**作用**：启动时将项目根目录的 MP4 演示视频同步到 `uploads/videos/`，供前端 `<video>` 播放。

```javascript
function prepareDemoVideos() {
  removeLegacyDemoFiles(); // 清理旧版 wmv 演示文件

  for (const item of DEMO_VIDEO_FILES) {
    const source = findSourceFile(item); // 查找 9.2-1 / 9.4-2 等源 MP4
    if (!source) continue;

    const mp4Dest = path.join(UPLOADS_DIR, `${item.storageKey}.mp4`);
    ensureMp4(source, mp4Dest); // 复制到统一路径，便于静态资源访问
  }
}

function getDemoVideos() {
  return DEMO_VIDEO_FILES.map((item) => {
    const mp4Path = path.join(UPLOADS_DIR, `${item.storageKey}.mp4`);
    const hasMp4 = fs.existsSync(mp4Path) && fs.statSync(mp4Path).size > 0;
    return {
      id: item.id,
      title: item.title,
      url: hasMp4 ? `/uploads/videos/${item.storageKey}.mp4` : null,
      available: hasMp4,
      playable: hasMp4,
    };
  });
}
```

### 6. 前端 HTTP 拦截器 — Token 注入与统一错误处理

**文件**：`frontend/src/api/request.js`  
**作用**：自动携带 JWT；401 时清除登录态并跳转登录页。

```javascript
const request = axios.create({ baseURL: '/api/v1', timeout: 15000 });

// 请求拦截：从 localStorage 读取 token 写入 Authorization 头
request.interceptors.request.use((config) => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截：统一处理业务 code 与网络/认证错误
request.interceptors.response.use(
  (response) => {
    const res = response.data;
    if (res.code !== 0) {
      ElMessage.error(res.message || '请求失败');
      return Promise.reject(new Error(res.message));
    }
    return res;
  },
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      localStorage.removeItem('user');
      router.push('/login'); // Token 失效，强制重新登录
    }
    ElMessage.error(/* 后端不可达时给出友好提示 */);
    return Promise.reject(error);
  }
);
```

### 7. 前端路由守卫 — 登录与管理员权限

**文件**：`frontend/src/router/index.js`  
**作用**：未登录跳转登录页；非 admin 不可访问 `/admin/*`。

```javascript
router.beforeEach((to) => {
  const userStore = useUserStore();

  // 非公开页且未登录 → 跳转登录
  if (!to.meta.public && !userStore.isLoggedIn) {
    return '/login';
  }

  // 路由 meta.roles 限定 admin，成员访问管理页则重定向首页
  if (to.meta.roles && !to.meta.roles.includes(userStore.user?.role)) {
    return '/dashboard';
  }

  // 已登录用户访问登录页 → 直接进入系统
  if (to.meta.public && userStore.isLoggedIn && to.name === 'Login') {
    return '/dashboard';
  }
});
```

## 三、不足之处与改进办法

| 不足 | 说明 | 改进办法 |
|------|------|----------|
| **无分页** | 列表接口一次返回全量数据 | 增加 `page`/`pageSize` 与 SQL `OFFSET-FETCH` |
| **JWT 无刷新** | Token 过期需重新登录 | 引入 Refresh Token 双令牌机制 |
| **密码找回无邮件** | 依赖管理员人工审核 | 接入 SMTP/企业微信通知 |
| **视频仅 MP4** | 不支持在线转码 | 上传后异步队列转码（ffmpeg + 任务表） |
| **无自动化测试** | 依赖手工测试 | 引入 Vitest + Supertest 覆盖核心 API |
| **单库单实例** | 未做读写分离与缓存 | 读多写少模块加 Redis 缓存 |
| **启动时视频同步** | 曾出现文件占用导致后端崩溃 | 已改为容错；可改为独立脚本，不在每次启动执行 |
| **Windows 环境依赖** | npm/sqlcmd 路径问题较多 | 提供 Docker Compose 一键部署 |
| **安全加固不足** | 实验环境配置较宽松 | 生产环境启用 HTTPS、限流、强密码策略与日志审计 |
